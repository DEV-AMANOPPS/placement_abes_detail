// Most asked pattern: Sliding Window — MUST KNOW!
int maxSubarrayLen(vector<int>& nums, int k) {
  int left = 0, sum = 0, maxLen = 0;
  for (int right = 0; right < nums.size(); right++) {
    sum += nums[right];
    while (sum > k) sum -= nums[left++];
    maxLen = max(maxLen, right - left + 1);
  }
  return maxLen;
}

SQL — Must-Know Queries ▼
-- 2nd Highest Salary (bahut common!)
SELECT MAX(salary) FROM employees
WHERE salary < (SELECT MAX(salary) FROM employees);

-- Department wise highest salary
SELECT dept, MAX(salary) FROM employees GROUP BY dept;

-- Employees joined in last 30 days
SELECT * FROM employees
WHERE join_date >= CURDATE() - INTERVAL 30 DAY;

-- Count employees per department having > 5
SELECT dept, COUNT(*) FROM employees
GROUP BY dept HAVING COUNT(*) > 5;
